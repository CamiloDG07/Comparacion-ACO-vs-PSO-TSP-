\begin{itemize}
\item \textbf{Calidad: ACO es mejor.} Con 2\,048 agentes ACO obtiene menor $L_{mejor}$ en 15 de 15 combinaciones de $n$ y presupuesto; en las 9 configuraciones emparejadas de cada $n\le 20\,000$ gana en 9/9 ($n=20$), 9/9 ($n=200$), 9/9 ($n=2\,000$), 9/9 ($n=20\,000$). Razón mediana $L_{PSO}/L_{ACO}$: $n=20$: 1,32, $n=200$: 7,63, $n=2\,000$: 26,33, $n=20\,000$: 84,42.
\item \textbf{Dependencia de $n$: no hay cruce en el rango probado ($n=20$ a $n=200\,000$).} Razón $L_{PSO}/L_{ACO}$ con 2\,048 agentes y 100 it. (108 en $n=200\,000$): $n=20$: 1,05, $n=200$: 7,23, $n=2\,000$: 26,73, $n=20\,000$: 84,80, $n=200\,000$: 269,03.
\item \textbf{Presupuesto.} Pasar de 5 a 100 iteraciones (de 5 a 20 en $n=200\,000$) mejora $L_{mejor}$, con 2\,048 agentes, en ACO 0,0\,\% ($n=20$), 2,5\,\% ($n=200$), 3,0\,\% ($n=2\,000$), 1,8\,\% ($n=20\,000$), 1,0\,\% ($n=200\,000$) y en PSO 31,8\,\% ($n=20$), 8,4\,\% ($n=200$), 0,7\,\% ($n=2\,000$), 0,3\,\% ($n=20\,000$), 0,0\,\% ($n=200\,000$).
\item \textbf{Tiempo: PSO es más rápido en todas las configuraciones emparejadas.} Razón $t_{PSO}/t_{ACO}$ mediana en las 9 configuraciones: 0,47 (rango 0,39--0,50, PSO más rápido en 9/9; $n=20$), 0,43 (rango 0,40--0,51, PSO más rápido en 9/9; $n=200$), 0,50 (rango 0,45--0,53, PSO más rápido en 9/9; $n=2\,000$), 0,60 (rango 0,55--0,68, PSO más rápido en 9/9; $n=20\,000$); con $n=200\,000$, 2\,048 agentes y 108 it.: 0,63.
\item \textbf{Memoria: PSO usa más en todo el rango.} Razón $M_{PSO}/M_{ACO}$ con 2\,048 agentes: 1,1 ($n=20$), 2,0 ($n=200$), 9,2 ($n=2\,000$), 33,3 ($n=20\,000$), 46,2 ($n=200\,000$).
\item \textbf{A igual tiempo:} $n=20$ ($T=0,07$ s): $L/L_{ref}$ ACO 1,000, PSO 1,050; $n=200$ ($T=0,32$ s): $L/L_{ref}$ ACO 1,009, PSO 7,236; $n=2\,000$ ($T=3,53$ s): $L/L_{ref}$ ACO 1,010, PSO 26,806; $n=20\,000$ ($T=42,38$ s): $L/L_{ref}$ ACO 1,001, PSO 84,844; $n=200\,000$ ($T=610,08$ s): $L/L_{ref}$ ACO 1,002, PSO 269,031.
\item \emph{Límites:} PSO con parámetros fijos ($w=0{,}7$, $c_1=c_2=1{,}5$) y sin búsqueda local; los resultados valen para esta implementación.
\end{itemize}
